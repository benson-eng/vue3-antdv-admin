    title: '角色名稱',

    title: '角色值',
    width: 180,
    dataIndex: 'value',
  },
  {
    title: '狀態',
    dataIndex: 'status',
    width: 80,
    // 搜索表單配置
    formItemProps: {
      component: 'Select',
      componentProps: {
        options: [
          { label: '啟用', value: 1 },
          { label: '禁用', value: 0 },
        ],
      },
    },
    // 自定義渲染
    customRender: ({ record }) => {
      const enable = ~~record.status === 1;
      return <Tag color={enable ? 'green' : 'red'}>

      </Tag>;
    },

    title: '創建時間',

#### 列配置屬性說明

| 屬性 | 類型 | 說明 |

| `actions` | `function` | 操作列配置（返回 ActionItem[]） |

### 高級功能

#### 1. 批量操作

```vue
<template>
  <DynamicTable
    :data-request="loadTableData"
    :columns="columns"
    :row-selection="rowSelection"
  >
    <template #toolbar>
      <a-button 
        type="error" 
        :disabled="!isCheckRows"
        @click="handleBatchDelete"
      >
        批量刪除
      </a-button>
    </template>
  </DynamicTable>
</template>

<script setup lang="ts">
const rowSelection = ref({
  selectedRowKeys: [] as number[],
  onChange: (selectedRowKeys: number[]) => {
    rowSelection.value.selectedRowKeys = selectedRowKeys;
  },
});

const isCheckRows = computed(() => {
  return rowSelection.value.selectedRowKeys.length > 0;
});

const handleBatchDelete = async () => {
  const ids = rowSelection.value.selectedRowKeys;
  await Api.module.batchDelete({ ids });
  dynamicTableInstance?.reload();
  rowSelection.value.selectedRowKeys = [];
};
</script>
```

## Modal 彈窗管理

### useFormModal Hook

#### 基本用法

```typescript
import { useFormModal } from '@/hooks/useModal/';
import { roleSchemas } from './formSchemas';

const [showModal] = useFormModal();

const openModal = async (record: Partial<TableListItem> = {}) => {
  const isUpdate = Boolean(record.id);
  
  const [formRef] = await showModal({
    modalProps: {
      title: `${isUpdate ? '編輯' : '新增'}角色`,
      width: '50%',
      onFinish: async (values) => {
        // 提交表單
        if (isUpdate) {
          await Api.systemRole.roleUpdate({ id: record.id }, values);
        } else {
          await Api.systemRole.roleCreate(values);
        }
        
        // 刷新表格
        dynamicTableInstance?.reload();
      },
    },
    formProps: {
      labelWidth: 100,
      schemas: roleSchemas,
    },
  });
  
  // 編輯模式：回填數據
  if (isUpdate) {
    const roleInfo = await Api.systemRole.roleInfo({ id: record.id });
    formRef?.setFieldsValue(roleInfo);
  }
};
```

### 彈窗配置說明

#### modalProps 配置

| 屬性 | 類型 | 說明 |
|------|------|------|
| `title` | `string` | 彈窗標題 |
| `width` | `string \| number` | 彈窗寬度 |
| `onFinish` | `function` | 表單提交成功回調 |
| `onCancel` | `function` | 取消回調 |

#### formProps 配置

| 屬性 | 類型 | 說明 |
|------|------|------|
| `labelWidth` | `number` | 標籤寬度 |
| `schemas` | `FormSchema[]` | 表單配置 |
| `labelCol` | `object` | 標籤柵格配置 |
| `wrapperCol` | `object` | 控件柵格配置 |

### 完整範例

```typescript
const openUserModal = async (record: Partial<TableListItem> = {}) => {
  const isUpdate = Boolean(record.id);

  const [formRef] = await showModal({
    modalProps: {
      title: `${isUpdate ? '編輯' : '新增'}用戶`,
      width: 700,
      onFinish: async (values) => {
        // 處理數據
        const params = {
          ...values,
          roleIds: values.roleIds?.join(','),
        };
        
        // 調用 API
        if (isUpdate) {
          await Api.systemUser.userUpdate({ id: record.id }, params);
        } else {
          await Api.systemUser.userCreate(params);
        }
        
        // 刷新表格
        dynamicTableInstance?.reload();
      },
    },
    formProps: {
      labelWidth: 100,
      schemas: userSchemas,
    },
  });

  // 加載下拉選項數據
  const [roles, depts] = await Promise.all([
    Api.systemRole.roleList({}),
    Api.systemDept.deptList({}),
  ]);

  // 更新表單配置
  formRef?.updateSchema([
    {
      field: 'roleIds',
      componentProps: {
        options: roles.map(r => ({ label: r.name, value: r.id })),
      },
    },
    {
      field: 'deptId',
      componentProps: {
        treeData: depts,
      },
    },
  ]);

  // 編輯模式：回填數據
  if (isUpdate) {
    const userInfo = await Api.systemUser.userInfo({ id: record.id });
    formRef?.setFieldsValue({
      ...userInfo,
      roleIds: userInfo.roleIds?.split(',').map(Number),
    });
  }
};
```

## 總結

### 核心設計原則

1. **組件化設計** - 使用 DynamicTable、SchemaForm 等核心組件提升開發效率
2. **配置驅動** - 通過 columns.tsx 和 formSchemas.ts 配置頁面，減少模板代碼
3. **類型安全** - 充分利用 TypeScript 提供類型檢查和智能提示
4. **權限控制** - 多層級權限控制確保功能安全
5. **統一規範** - 統一的代碼結構和命名規範提升可維護性

### 開發流程

```
1. 創建頁面結構
   └─ index.vue + columns.tsx + formSchemas.ts

2. 配置表格
   └─ 定義 columns，配置搜索項和操作列

3. 配置表單
   └─ 定義 formSchemas，設置驗證規則

4. 實現業務邏輯
   └─ 新增、編輯、刪除等 CRUD 操作

5. 添加權限控制
   └─ 按鈕、操作列權限配置

6. 樣式調整
   └─ 使用 scoped 樣式和 CSS 變量
```
